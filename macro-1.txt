/**
 * Queen of Storms — Lifepath Wizard
 * Foundry VTT V14 · dnd5e 5.3.3 · ApplicationV2
 *
 * Implements John Wick's "Backstory" lifepath. Select one token, run the macro,
 * walk the six parts (+ optional trait pairs), and write the assembled result
 * into the character's Biography (system.details.biography.value).
 *
 * Option pools live in the compendium pack
 *   cv-queen-of-storms-compendiums.qos-lifepath-tables   (one RollTable per pool).
 * Run the companion "Seed Lifepath Tables" macro once to build/refresh them.
 * Branch routing (1B / 1E / Table 3) stays in this macro; only leaf pools are tables.

(async () => {
  // ---- 0. Preconditions --------------------------------------------------
  const AppV2 = foundry.applications?.api?.ApplicationV2;

    ui.notifications.warn("Select exactly one token, then run the Lifepath Wizard.");
    return;

  // flag key (so tables can be freely renamed in the UI). Branch routing

    return;

    const docs = await pack.getDocuments();
    tableDocs = {};
    for (const t of docs) {
      const key = t.getFlag(MODULE_ID, "key");
      if (key) tableDocs[key] = t;

    console.error("Lifepath Wizard | failed to load tables", err);
    ui.notifications.error("Lifepath Wizard: could not read the lifepath tables compendium (see console).");

  const ordinal = (n) => { const s = ["th", "st", "nd", "rd"], v = n % 100; return n + (s[(v - 20) % 10] || s[v] || s[0]); };

  // ---- 5. The wizard -----------------------------------------------------
  class LifepathWizard extends AppV2 {
    constructor(targetActor, tables, options = {}) {
      super({ id: `qos-lifepath-${targetActor.id}`, ...options });
      this.actor = targetActor;
      this.tableText = tables.text;
      this.tableWeights = tables.weights;
      this.tableDocs = tables.docs;
      this.step = 1;
      this.saveResult = null;
      const existing = (targetActor.system?.details?.biography?.value || "").trim();
      this.data = {
        // Part 1
        status: "", childhood: "",
        parentsBranch: "", parentalDisaster: "", familySecret: "",
        standingBranch: "", familyCrisis: "",
        familyGoal: "", siblings: [], birthOrder: 0, siblingsUnknown: false,
        motherBond: "", fatherBond: "",
        // Part 2
        friends: [], enemies: [],
        // Part 3
        romanceStatus: "", romanceDetail: "",
        loverRace: "", loverProfession: "", loverAppearance: "",
        // Part 4
        hairColor: "", hairStyle: "", eyeColor: "", personalStyle: "",
        // Part 5
        trait: "", values: "", valuedObject: "", lostIt: "", fear: "", valuedPerson: "", quirk: "",
        // Optional trait pairs
        useTraitPairs: false, traitMode: "selected", traitPairs: {},
        // Part 6
        lifeEvents: [],
        // Save behaviour
        appendBio: !!existing,
      };
      for (let i = 0; i < TRAIT_PAIRS.length; i++) this.data.traitPairs[i] = 10;
    }

    static DEFAULT_OPTIONS = {
      id: "qos-lifepath",
      classes: ["qos-lifepath-wizard"],
      tag: "div",
      window: { title: "Lifepath Wizard", icon: "fa-solid fa-scroll", resizable: true },
      position: { width: 900, height: "auto" },
    };

    get title() { return `Lifepath Wizard — ${this.actor?.name ?? ""}`; }

    // ===== render =====
    async _renderHTML() {
      injectStyles();
      const s = this.step;
      const pct = Math.round((s / STEP_COUNT) * 100);
      const header = `
        <header class="qbw-header">
          <div class="qbw-step-no">Step ${s} of ${STEP_COUNT}</div>
          <h2 class="qbw-title">${STEP_TITLES[s]}</h2>
          <div class="qbw-progress"><div class="qbw-progress-bar" style="width:${pct}%"></div></div>
        </header>`;
      return `<div class="qbw">${header}
        <div class="qbw-main">
          <section class="qbw-body">${this._renderBody()}</section>
          <aside class="qbw-aside">${this._renderAside()}</aside>
        </div>
        ${this._renderFooter()}</div>`;
    }
    _replaceHTML(result, content) {
      // Preserve scroll position of both panes across full re-renders (Roll
      // buttons, branch changes, etc. rebuild innerHTML, which would reset them).
      const SCROLLERS = [".qbw-body", ".qbw-aside"];
      const tops = {};
      for (const sel of SCROLLERS) { const p = content.querySelector(sel); if (p) tops[sel] = p.scrollTop; }
      content.innerHTML = result;
      for (const sel of SCROLLERS) { const n = content.querySelector(sel); if (n && tops[sel]) n.scrollTop = tops[sel]; }
    }

    // --- generic field renderers ---
    _tableRow(field, label, table) {
      const list = this.tableText[table] || [];
      const cur = this.data[field];
      const opts = ['<option value="">— choose or roll —</option>'].concat(
        list.map((txt, i) => `<option value="${esc(txt)}" ${txt === cur ? "selected" : ""}>${i + 1}. ${esc(txt)}</option>`)
      ).join("");
      return `<label class="qbw-label">${esc(label)}</label>
        <div class="qbw-row">
          <select class="qbw-select" data-field="${field}">${opts}</select>
          <button type="button" class="qbw-btn" data-act="roll" data-field="${field}" data-table="${table}"><i class="fa-solid fa-dice-d20"></i> Roll</button>
        </div>`;
    }
    _branchRow(field, name, label) {
      const b = BRANCH[name];
      const cur = this.data[field];
      const opts = ['<option value="">— choose or roll —</option>'].concat(
        b.options.map((o) => `<option value="${o.value}" ${o.value === cur ? "selected" : ""}>${esc(o.label)}</option>`)
      ).join("");
      return `<label class="qbw-label">${esc(label)}</label>
        <div class="qbw-row">
          <select class="qbw-select" data-field="${field}" data-rerender="1">${opts}</select>
          <button type="button" class="qbw-btn" data-act="rollbranch" data-field="${field}" data-branch="${name}"><i class="fa-solid fa-dice-d20"></i> Roll d${b.die}</button>
        </div>`;
    }

    _renderBody() {
      switch (this.step) {
        case 1: return this._renderFamily();
        case 2: return this._renderFriendsEnemies();
        case 3: return this._renderRomance();
        case 4: return this._renderAppearance();
        case 5: return this._renderPersonality();
        case 6: return this._renderLifeEvents();
        case 7: return this._renderTraitPairs();
        default: return "";
      }
    }

    _renderFamily() {
      const d = this.data;
      let h = `<div class="qbw-section-title">Family &amp; Money</div>`;
      h += this._tableRow("status", "Social standing (Table 1)", "status");
      h += this._tableRow("childhood", "Childhood event (1A)", "childhood");
      h += `<div class="qbw-section-title">Parents</div>`;
      h += this._branchRow("parentsBranch", "parents", "Family situation (1B)");
      if (d.parentsBranch === "disaster") h += this._tableRow("parentalDisaster", "What happened to your parents? (1C)", "parentalDisaster");
      else if (d.parentsBranch === "special") h += this._tableRow("familySecret", "Your family's secret (1D)", "familySecret");
      else if (d.parentsBranch === "alive") h += `<div class="qbw-hint">Both parents are alive and well — no further roll.</div>`;
      h += this._tableRow("motherBond", "Bond with your mother (0R)", "familyBond");
      h += this._tableRow("fatherBond", "Bond with your father (0R)", "familyBond");
      h += `<div class="qbw-hint">A bond is kept for each parent whether they are living or not.</div>`;
      h += `<div class="qbw-section-title">Standing &amp; Legacy</div>`;
      h += this._branchRow("standingBranch", "standing", "Family standing (1E)");
      if (d.standingBranch === "bad") h += this._tableRow("familyCrisis", "Current family crisis (1F)", "familyCrisis");
      else if (d.standingBranch === "good") h += `<div class="qbw-hint">Your family status is good — no crisis.</div>`;
      h += this._tableRow("familyGoal", "Family goal (1G)", "familyGoal");
      h += this._renderSiblings();
      return h;
    }

    _listCard(listKey, idx, title, fields) {
      const entry = this.data[listKey][idx];
      let body = "";
      for (const [sub, label, table] of fields) {

    // ===== listeners (delegated) =====
    // Bound on EVERY render via _onRender, guarded so we attach exactly once
    // per root element. AppV2 can replace this.element on re-render (e.g. when
    // position height is "auto"); binding only in _onFirstRender would leave
    // later roots without listeners, so user input would stop reaching state
    // and the next render would blank every field. Re-binding fixes that.
    _onRender(context, options) {
      super._onRender?.(context, options);
      const el = this.element;
      if (!el || el.__qbwBound) return;
      el.__qbwBound = true;
      el.addEventListener("click", (ev) => {
        const btn = ev.target.closest("[data-act]");
        if (!btn || !el.contains(btn)) return;
        ev.preventDefault();
        this._handleAction(btn.dataset.act, btn);
      });
      el.addEventListener("change", (ev) => {
        const t = ev.target;
        // Capture every visible control into state first, so a re-render never
        // discards selections whose change events we might otherwise miss.
        this._syncFromDom();
        if (t.dataset.list && t.dataset.idx !== undefined && t.dataset.sub) {
          const list = this.data[t.dataset.list]; const i = +t.dataset.idx;
          if (list && list[i]) {
            this._touch();
            if (t.dataset.sub === "luck") { list[i].text = ""; this.render(); return; }
            if (t.dataset.rerender) { this.render(); return; }
          }
          this._refreshPreview();
          return;
        }
        if (t.dataset.field) {
          this._touch();
          if (t.dataset.field === "romanceStatus") this.data.romanceDetail = "";
          if (t.dataset.rerender) { this.render(); return; }
        }
        this._refreshPreview();
      });
      el.addEventListener("input", (ev) => {
        const t = ev.target;
        if (t.dataset.pair !== undefined) {
          const i = +t.dataset.pair;
          const left = Math.max(0, Math.min(20, parseInt(t.value, 10) || 0));
          this.data.traitPairs[i] = left;
          this._touch();
          const ls = el.querySelector(`[data-pairleft="${i}"]`); if (ls) ls.textContent = left;
          const rs = el.querySelector(`[data-pairright="${i}"]`); if (rs) rs.textContent = 20 - left;
          const bd = el.querySelector("[data-pairbudget]"); if (bd) bd.textContent = this._pairBudget();
          if (this.data.useTraitPairs) this._refreshPreview();
        }
      });
    }

    // ===== state mutation =====
    _touch() { this.saveResult = null; }

    _syncFromDom() {
      const el = this.element; if (!el) return;
      // Only value-bearing controls. Roll buttons also carry data-field (for the
      // roll handler), but a button's value is "" and would clobber its select.
      el.querySelectorAll("select[data-field], input[data-field]").forEach((f) => {
        if (f.type === "checkbox") this.data[f.dataset.field] = f.checked;
        else this.data[f.dataset.field] = f.value;
      });
      el.querySelectorAll("[data-list][data-idx][data-sub]").forEach((f) => {
        const list = this.data[f.dataset.list]; const i = +f.dataset.idx;
        if (list && list[i]) list[i][f.dataset.sub] = (f.type === "checkbox") ? f.checked : f.value;
      });
      el.querySelectorAll("[data-pair]").forEach((f) => {
        this.data.traitPairs[+f.dataset.pair] = Math.max(0, Math.min(20, parseInt(f.value, 10) || 0));
      });
    }

    _rollBranch(name) {
      const b = BRANCH[name];
      const r = randInt(1, b.die);
      return (b.options.find((o) => r >= o.lo && r <= o.hi) ?? b.options[0]).value;
    }
    // 1d100 -> the character's birth position (1 = eldest ... 6 = sixth child).
    _rollBirthOrder() {
      const r = randInt(1, 100);
      return (BIRTH_ORDER.find((o) => r >= o.lo && r <= o.hi) ?? BIRTH_ORDER[0]).n;
    }
    // 1d100 -> whether the character has siblings, and whether that is certain.
    _rollHaveSiblings() {
      const r = randInt(1, 100);
      const none = r >= SIBLINGS_NONE_RANGE[0] && r <= SIBLINGS_NONE_RANGE[1];
      return { has: !none, unknown: r >= SIBLINGS_UNKNOWN_MIN };
    }
    // Derive a sibling's kind from its current position: the siblings immediately
    // before/after the character roll for twin (of the character); everyone else
    // rolls for half-sibling. Using the stored rolls keeps this consistent if the
    // character's position changes, and never marks a twin as a half-sibling.
    _siblingKind(s, position, X) {
      const adjacent = position === X - 1 || position === X + 1;
      if (adjacent) {
        const r = s.twinRoll || 0;
        if (r >= TWIN_IDENTICAL_MIN) return { key: "identical", label: "identical twin" };
        if (r >= TWIN_FRATERNAL_MIN) return { key: "fraternal", label: "fraternal twin" };
        return { key: "full", label: "" };
      }
      return (s.halfRoll || 0) >= HALF_SIBLING_MIN ? { key: "half", label: "half-sibling" } : { key: "full", label: "" };
    }
    // The character's sex from the sheet's free-text gender field; anything that
    // isn't clearly male/female defaults to female. Used to keep an identical
    // twin the same sex as the character.
    _pcSex() {
      const g = String(this.actor?.system?.details?.gender || "").trim().toLowerCase();
      if (g === "male") return "male";
      if (g === "female") return "female";
      return "female";
    }
    // Weighted pick over a cached pool (weight = range span). Resolves the same
    // distribution RollTable#roll would, but synchronously and silently — no
    // chat card, no Dice So Nice — to keep the biographical, instant feel.
    _rollKey(key) {
      const names = this.tableText[key] || [];
      if (!names.length) return "";
      const weights = this.tableWeights[key] || [];
      let total = 0;
      for (let i = 0; i < names.length; i++) total += weights[i] || 1;
      let roll = Math.random() * total;
      for (let i = 0; i < names.length; i++) { roll -= (weights[i] || 1); if (roll < 0) return names[i]; }
      return names[names.length - 1];
    }
    _randEntry(list) {
      if (list === "friends") return { situation: this._rollKey("friendSituation"), race: this._rollKey("feRace"), profession: this._rollKey("feProfession") };
      if (list === "enemies") return { situation: this._rollKey("enemySituation"), race: this._rollKey("feRace"), profession: this._rollKey("feProfession") };
      if (list === "siblings") return { relation: this._rollKey("siblings"), bond: this._rollKey("familyBond"), sex: randInt(1, 2) === 1 ? "male" : "female", alive: randInt(1, 100) > SIBLING_DEATH_PCT, twinRoll: randInt(1, 100), halfRoll: randInt(1, 100) };
      if (list === "lifeEvents") { const luck = randInt(1, 6) <= 3 ? "unlucky" : "lucky"; return { luck, text: this._rollKey(luck) }; }
      return {};
    }
    _regenList(list, n) {
      this._syncFromDom();
      const arr = [];
      for (let k = 0; k < n; k++) arr.push(this._randEntry(list));
      this.data[list] = arr;
      this._touch();
    }

    _handleAction(act, btn) {
      switch (act) {
        case "next": this._goStep(this.step + 1); break;
        case "back": this._goStep(this.step - 1); break;
        case "close": this.close(); break;
        case "save": this._save(); break;

        case "roll": {
          this._syncFromDom();
          const v = this._rollKey(btn.dataset.table);
          if (v) this._setField(btn.dataset.field, v);
          this.render(); break;
        }
        case "rollbranch": {
          this._syncFromDom();
          const field = btn.dataset.field;
          this._setField(field, this._rollBranch(btn.dataset.branch));
          if (field === "romanceStatus") this.data.romanceDetail = "";
          this.render(); break;
        }
        case "rollcount": this._regenList(btn.dataset.list, randInt(1, +btn.dataset.die)); this.render(); break;
        case "roll-family": {
          this._syncFromDom();
          const { has, unknown } = this._rollHaveSiblings();
          if (!has) {                                    // 76-90: only child
            this.data.birthOrder = 1;
            this.data.siblings = [];
            this.data.siblingsUnknown = false;
          } else if (unknown) {                          // 91-100: "...as far as you know" — none detailed
            this.data.birthOrder = 0;
            this.data.siblings = [];
            this.data.siblingsUnknown = true;
          } else {                                       // 01-75: known siblings
            const X = this._rollBirthOrder();
            let younger = randInt(1, 6) - 1;            // 0-5 younger siblings
            if (X === 1 && younger === 0) younger = 1;   // guarantee at least one
            const Y = X + younger;
            this.data.birthOrder = X;
            this.data.siblingsUnknown = false;
            const arr = [];
            for (let k = 0; k < Y - 1; k++) arr.push(this._randEntry("siblings"));
            this.data.siblings = arr;
          }
          this._touch(); this.render(); break;
        }
        case "roll-sib": {
          this._syncFromDom();
          const i = +btn.dataset.idx;
          const sub = btn.dataset.sub || "relation";
          const table = btn.dataset.table || "siblings";
          if (this.data.siblings[i]) this.data.siblings[i][sub] = this._rollKey(table);
          this._touch(); this.render(); break;
        }
        case "gen": {
          const list = btn.dataset.list;
          const input = this.element.querySelector(`[data-countfield="${list}"]`);
          let n = parseInt(input?.value, 10);
          if (!Number.isFinite(n) || n < 0) n = this.data[list].length;
          this._regenList(list, Math.max(0, Math.min(20, n)));
          this.render(); break;
        }
        case "add": this._syncFromDom(); this.data[btn.dataset.list].push(this._randEntry(btn.dataset.list)); this._touch(); this.render(); break;
        case "reroll-entry": this._syncFromDom(); this.data[btn.dataset.list][+btn.dataset.idx] = this._randEntry(btn.dataset.list); this._touch(); this.render(); break;
        case "remove-entry": this._syncFromDom(); this.data[btn.dataset.list].splice(+btn.dataset.idx, 1); this._touch(); this.render(); break;

        case "pairs-mode": this._syncFromDom(); this.data.traitMode = btn.dataset.mode; this.render(); break;
        case "pairs-roll": for (let i = 0; i < TRAIT_PAIRS.length; i++) this.data.traitPairs[i] = randInt(1, 6) + randInt(1, 6) + randInt(1, 6); this._touch(); this.render(); break;
        case "pairs-reset": for (let i = 0; i < TRAIT_PAIRS.length; i++) this.data.traitPairs[i] = 10; this._touch(); this.render(); break;
      }
    }

    // ===== output =====
    _buildBiographyHtml() {
      const d = this.data;
      const P = (label, val) => val ? `<p><strong>${label}:</strong> ${esc(val)}</p>` : "";
      let out = "";

      // Family
      let fam = P("Social standing", d.status) + P("Childhood", d.childhood);
      if (d.parentsBranch === "disaster") fam += `<p><strong>Parents:</strong> ${d.parentalDisaster ? esc(d.parentalDisaster) : "A disaster befell one or both parents."}</p>`;
      else if (d.parentsBranch === "special") fam += `<p><strong>Parents:</strong> ${d.familySecret ? "Family secret — " + esc(d.familySecret) : "There is something special about them."}</p>`;
      else if (d.parentsBranch === "alive") fam += `<p><strong>Parents:</strong> Both alive and well.</p>`;
      if (d.standingBranch === "bad") fam += `<p><strong>Family standing:</strong> At risk — ${d.familyCrisis ? esc(d.familyCrisis) : "a crisis looms."}</p>`;
      else if (d.standingBranch === "good") fam += `<p><strong>Family standing:</strong> Secure, even amid hardship.</p>`;
      fam += P("Family goal", d.familyGoal);
      const bo = Number(d.birthOrder) || 0;
      if (d.siblingsUnknown && d.siblings.length === 0) {
        fam += `<p><strong>Siblings:</strong> Uncertain — you may have siblings, as far as you know.</p>`;
      } else if (bo >= 1) {
        const Y = d.siblings.length + 1;
        if (Y === 1) {
          fam += `<p><strong>Siblings:</strong> An only child.</p>`;
        } else {
          fam += `<p><strong>Birth order:</strong> ${ordinal(Math.min(bo, Y))} of ${Y} children.</p>`;
          const positions = [];
          for (let p = 1; p <= Y; p++) if (p !== Math.min(bo, Y)) positions.push(p);
          const X = Math.min(bo, Y);
          const pcSex = this._pcSex();
          const items = d.siblings.map((s, i) => {
            const pos = positions[i];
            const ord = pos < X ? "older" : "younger";
            const kind = this._siblingKind(s, pos, X);
            const sex = kind.key === "identical" ? pcSex : s.sex;
            const noun = sex === "male" ? "brother" : sex === "female" ? "sister" : "";
            const who = noun ? `${ord} ${noun}` : ord;
            const kindStr = kind.label ? `, ${kind.label}` : "";
            const rel = s.relation ? `: ${esc(s.relation)}` : "";
            const bond = s.bond ? ` (${esc(s.bond)})` : "";
            const dead = s.alive === false ? " — <em>deceased</em>" : "";
            return `<li>${ordinal(pos)} child (${who}${kindStr})${rel}${bond}${dead}</li>`;
          }).join("");
          fam += `<ul>${items}</ul>`;
        }
      }
      fam += P("Bond with mother", d.motherBond) + P("Bond with father", d.fatherBond);
      if (fam) out += "<h2>Family</h2>" + fam;

      // Friends & Enemies
      const liEntry = (e) => {
        const tail = [e.race, e.profession].filter(Boolean).join(", ");
        return `<li>${esc(e.situation || "")}${tail ? ` <em>(${esc(tail)})</em>` : ""}</li>`;
      };
      let fe = "";
      if (d.friends.length) fe += "<h3>Friends</h3><ul>" + d.friends.map(liEntry).join("") + "</ul>";
      if (d.enemies.length) fe += "<h3>Enemies</h3><ul>" + d.enemies.map(liEntry).join("") + "</ul>";
      if (fe) out += "<h2>Friends &amp; Enemies</h2>" + fe;

      // Romance
      if (d.romanceStatus) {
        let rom = `<p><strong>Romance:</strong> ${esc(ROMANCE_LABEL[d.romanceStatus])}.</p>`;
        if (d.romanceDetail) rom += `<p>${esc(d.romanceDetail)}</p>`;
        if (d.romanceStatus !== "lookout") {
          const tail = [d.loverRace, d.loverProfession].filter(Boolean).join(", ");
          if (tail || d.loverAppearance) {
            const who = d.romanceStatus === "tragic" ? "Lost lover" : "Lover";
            rom += `<p><strong>${who}:</strong> ${esc(tail)}${tail && d.loverAppearance ? ". " : ""}${esc(d.loverAppearance || "")}</p>`;
          }
        }
        out += "<h2>Romance</h2>" + rom;
      }

      // Appearance
      let app = P("Hair", [d.hairColor, d.hairStyle].filter(Boolean).join(", ")) + P("Eyes", d.eyeColor) + P("Personal style", d.personalStyle);
      if (app) out += "<h2>Appearance</h2>" + app;

      // Personality
      let per = P("Defining trait", d.trait) + P("Values most", d.values) + P("Most valued object", d.valuedObject)
        + P("If it were lost", d.lostIt) + P("Greatest fear", d.fear) + P("Person they value most", d.valuedPerson) + P("Quirk", d.quirk);
      if (d.useTraitPairs) {
        per += "<h3>Trait pairs</h3><ul>" + TRAIT_PAIRS.map((pr, i) => {
          const l = d.traitPairs[i] ?? 10; return `<li>${pr[0]} ${l} / ${pr[1]} ${20 - l}</li>`;
        }).join("") + "</ul>";
      }
      if (per) out += "<h2>Personality</h2>" + per;

      // Life events
      const evLines = d.lifeEvents.map((e) => e.text ? `<li><strong>${e.luck === "lucky" ? "Lucky" : "Unlucky"}:</strong> ${esc(e.text)}</li>` : "").join("");
      if (evLines) out += "<h2>Life Events</h2><ul>" + evLines + "</ul>";

      return out ? "<h1>Lifepath</h1>" + out : "";
    }

    async _save() {
      this._syncFromDom();
      const html = this._buildBiographyHtml();
      if (!html) {
        this.saveResult = { ok: false, msg: "Nothing to save yet — fill in at least one section." };
        ui.notifications.warn("Lifepath Wizard: nothing to save yet.");
        this.render(); return;
      }
      try {
        const existing = this.actor.system?.details?.biography?.value || "";
        const value = (this.data.appendBio && existing.trim())
          ? `${existing}\n<hr>\n${html}`
          : html;
        await this.actor.update({ "system.details.biography.value": value });
        this.saveResult = { ok: true, msg: `Lifepath written to ${this.actor.name}'s biography.` };
        ui.notifications.info(`Lifepath written to ${this.actor.name}'s biography.`);
      } catch (err) {
        console.error("Lifepath Wizard | save failed", err);
        this.saveResult = { ok: false, msg: `Save failed: ${err.message}` };
        ui.notifications.error(`Lifepath Wizard save failed: ${err.message}`);
      }
      this.render();
    }
  }

  // ---- 6. Launch ---------------------------------------------------------
  const id = `qos-lifepath-${actor.id}`;
  const existing = foundry.applications?.instances?.get?.(id);
  if (existing) existing.close();
  new LifepathWizard(actor, { text: tableText, weights: tableWeights, docs: tableDocs }).render({ force: true });
})();
